## Part 2: Coding challenge

Write a **calculator** for integer expressions such as `2 * (3 + -4) % 5`. The classic way is a **recursive-descent
parser**: one small method per grammar rule ([Methods](/lessons/methods-basics/)), calling each other recursively for
nested parentheses ([The Call Stack](/lessons/call-stack/)), with a depth limit so a hostile input gets a clean error
instead of a `StackOverflowError`. The public surface is one static method; the parsing state lives in a private
nested class, one instance per evaluation, so nothing is shared between calls or threads
([static vs Instance](/lessons/static-vs-instance/), [Packages & Access Modifiers](/lessons/packages-access-modifiers/)).

```java title="Calculator.java"
package track.checkpoint_3;

/**
 * 🔴 Checkpoint 3 challenge: a recursive-descent calculator.
 *
 * <p>Grammar (spaces allowed between tokens):
 * <pre>
 *   expression := term (('+' | '-') term)*
 *   term       := unary (('*' | '/' | '%') unary)*
 *   unary      := '-' unary | primary
 *   primary    := number | '(' expression ')'
 *   number     := decimal digits, 0 .. 2147483647
 * </pre>
 * So unary minus binds tightest, then * / %, then + -, all left-associative: the same precedence as in Java.
 * Arithmetic is Java's {@code int} arithmetic: overflow wraps, {@code /} truncates toward zero, {@code %} takes the
 * sign of the dividend, and division or remainder by zero throws {@link ArithmeticException}. (Unlike Java, a literal
 * can't be 2147483648, so write {@code -2147483647 - 1} for the smallest int; {@code - - 1} is 1.)
 *
 * <p>Errors throw {@link IllegalArgumentException} with a message ending in {@code "at N"}, N being the 0-based position
 * of the problem (the length of the input for an unexpected end):
 * <ul>
 *   <li>{@code "1 +"} → "unexpected end at 3"; {@code "(1 + 2"} → "expected ')' at 6"; {@code "1 2"} →
 *       "unexpected '2' at 2"; {@code "2147483648"} → "number too large at 0".</li>
```

```java title="CalculatorTest.java"
        assertThat(Calculator.evaluate("(1 + 2) * 3")).isEqualTo(9);

            {"2147483648", "number too large at 0"},
            {"1 + (2 * 3))", "unexpected ')' at 11"},

            assertThatThrownBy(() -> Calculator.evaluate(c[0])).as("\"%s\"", c[0])
                    .isInstanceOf(IllegalArgumentException.class).hasMessageEndingWith(c[1]);
        }

    @Test
    void deepNestingIsBoundedNotAStackOverflow() {
        assertThat(Calculator.evaluate("(".repeat(500) + "42" + ")".repeat(500))).isEqualTo(42);
        assertThatIllegalArgumentException().isThrownBy(() -> Calculator.evaluate("(".repeat(501) + "1" + ")".repeat(501)))
                .withMessageContaining("nested too deeply");
        assertThatIllegalArgumentException().isThrownBy(() -> Calculator.evaluate("(".repeat(100_000) + "1" + ")".repeat(100_000)))
                .withMessageContaining("nested too deeply");
        assertThatIllegalArgumentException().isThrownBy(() -> Calculator.evaluate("- ".repeat(100_000) + "1"))
                .withMessageContaining("nested too deeply");

        Random random = new Random(3_3);
        List<String> expressions = new ArrayList<>();

    @Test
    void isSafeToCallFromManyThreads() throws Exception {
        Random random = new Random(8);
        List<String> expressions = new ArrayList<>();
        List<Integer> answers = new ArrayList<>();
        while (expressions.size() < 300) {
            String e = expression(random, 5);
            try {
                answers.add(Calculator.evaluate(e));
                expressions.add(e);
            } catch (ArithmeticException ignored) {
                // skip divisions by zero
            }
        }
        ExecutorService pool = Executors.newFixedThreadPool(8);
        try {
            List<Future<Boolean>> results = new ArrayList<>();
            for (int t = 0; t < 8; t++) {
                results.add(pool.submit(() -> {
                    for (int round = 0; round < 50; round++) {
                        for (int i = 0; i < expressions.size(); i++) {
                            if (Calculator.evaluate(expressions.get(i)) != answers.get(i)) {
                                return false;
                            }
                        }
                    }
                    return true;
                }));
            }
            for (Future<Boolean> r : results) {
                assertThat(r.get()).isTrue();
            }
        } finally {
            pool.shutdownNow();
        }
        List<Class<?>> classes = new ArrayList<>(List.of(Calculator.class.getDeclaredClasses()));
        classes.add(Calculator.class);
        for (Class<?> c : classes) {
            for (Field f : c.getDeclaredFields()) {
                if (!f.isSynthetic() && Modifier.isStatic(f.getModifiers())) {
                    assertThat(Modifier.isFinal(f.getModifiers())).as("static field %s.%s must be final", c.getSimpleName(), f.getName()).isTrue();
                }
            }
        }
    }

    private static String expression(Random random, int depth) {
        if (depth == 0 || random.nextInt(4) == 0) {
            return String.valueOf(random.nextInt(3) == 0 ? EDGE[random.nextInt(EDGE.length)] : random.nextInt(0, 1000));
        }
        return switch (random.nextInt(6)) {
            case 0 -> "(" + expression(random, depth - 1) + ")";
            case 1 -> "(-" + expression(random, depth - 1) + ")";          // never "--", which Java reads as a decrement
            default -> expression(random, depth - 1) + " " + OPS[random.nextInt(OPS.length)] + " " + expression(random, depth - 1);
        };
    }
}
```

```java title="Calculator.java (solution)"
package track.checkpoint_3;

import java.util.Objects;

/** Reference solution for the Checkpoint 3 challenge. */
public final class Calculator {

    static final int MAX_DEPTH = 500;                  // a constant: final statics are fine

    private Calculator() {
    }

    public static int evaluate(String expression) {
        Objects.requireNonNull(expression, "expression");
        return new Parser(expression).parseAll();      // a fresh parser per call: nothing is shared
    }

    /** One evaluation. Its position and depth are instance state, owned by this call only. */
    private static final class Parser {
        private final String s;
        private int pos;
        private int depth;

        private Parser(String s) {
            this.s = s;
        }

        int parseAll() {
            int value = expression();
            skipSpaces();
            if (pos < s.length()) {
                throw error("unexpected '" + s.charAt(pos) + "'");
            }
            return value;
        }

        // expression := term (('+' | '-') term)*   -- the loop makes + and - left-associative
        private int expression() {
            int value = term();
            while (true) {
                skipSpaces();
                if (accept('+')) {
                    value += term();
                } else if (accept('-')) {
                    value -= term();
                } else {
                    return value;
                }
            }
        }

        // term := unary (('*' | '/' | '%') unary)*
        private int term() {
            int value = unary();
            while (true) {
                skipSpaces();
                if (accept('*')) {
                    value *= unary();
                } else if (accept('/')) {
                    value /= unary();                  // ArithmeticException on zero, as in Java
                } else if (accept('%')) {
                    value %= unary();
                } else {
                    return value;
                }
            }
        }

        // unary := '-' unary | primary
        private int unary() {
            skipSpaces();
            if (accept('-')) {
                enter();
                int value = -unary();
                depth--;
                return value;
            }
            return primary();
        }

        // primary := number | '(' expression ')'
        private int primary() {
            skipSpaces();
            if (accept('(')) {
                enter();
                int value = expression();
                skipSpaces();
                if (!accept(')')) {
                    throw error("expected ')'");
                }
                depth--;
                return value;
            }
            if (pos < s.length() && isDigit(s.charAt(pos))) {
                return number();
            }
            throw error(pos < s.length() ? "unexpected '" + s.charAt(pos) + "'" : "unexpected end");
        }

        private int number() {
            int start = pos;
            long value = 0;
            while (pos < s.length() && isDigit(s.charAt(pos))) {
                value = value * 10 + (s.charAt(pos) - '0');
                if (value > Integer.MAX_VALUE) {
                    throw new IllegalArgumentException("number too large at " + start);
                }
                pos++;
            }
            return (int) value;
        }

        /** Bounds the recursion, so deep input fails cleanly instead of exhausting the thread's stack. */
        private void enter() {
            if (++depth > MAX_DEPTH) {
                throw error("nested too deeply");
            }
        }

        private void skipSpaces() {
            while (pos < s.length() && s.charAt(pos) == ' ') {
                pos++;
            }
        }

        private boolean accept(char c) {
            if (pos < s.length() && s.charAt(pos) == c) {
                pos++;
                return true;
            }
            return false;
        }

        private static boolean isDigit(char c) {
            return c >= '0' && c <= '9';               // ASCII only: Character.isDigit accepts other scripts
        }

        private IllegalArgumentException error(String message) {
            return new IllegalArgumentException(message + " at " + pos);
        }
    }
}
```
